\section{Inferencia en LPO II}

\subsection{Encadenamiento hacia adelante (Forward Chaining)}

Aplica reglas de la forma $P_1 \land \cdots \land P_n \Rightarrow Q$ para derivar nuevos hechos hasta alcanzar la meta.
\begin{itemize}
    \item Completo para bases de conocimiento de \textbf{cláusulas de Horn}.
    \item Dirigido por los datos.
\end{itemize}

\subsection{Encadenamiento hacia atrás (Backward Chaining)}

Parte de la meta y trabaja hacia atrás buscando hechos o submetas.
\begin{itemize}
    \item Completo para cláusulas de Horn.
    \item Base de Prolog.
    \item Puede ciclar si no se controla.
\end{itemize}

\subsection{Resolución en LPO}

Generaliza la resolución proposicional usando unificación:

$$\frac{\ell_1 \lor \cdots \lor \ell_k \quad \neg m_1 \lor \cdots \lor \neg m_j}{\text{SUBST}(\theta, \ell_1 \lor \cdots \lor m_j)}$$

donde $\theta = \text{UNIFY}(\ell_i, m_k)$.

\begin{itemize}
    \item \textbf{Refutación por resolución}: completa para LPO (Teorema de Completitud de Gödel).
\end{itemize}

\subsection{Ejemplo: El caso de Nono (¿Es West un criminal?)}

\begin{example}[Base de conocimiento]
    \begin{enumerate}
        \item $American(x) \land Weapon(y) \land Sells(x,y,z) \land Hostile(z) \Rightarrow Criminal(x)$
        \item $Owns(Nono, M1)$
        \item $Missile(M1)$
        \item $Missile(x) \land Owns(Nono,x) \Rightarrow Sells(West,x,Nono)$
        \item $Missile(x) \Rightarrow Weapon(x)$
        \item $Enemy(x,America) \Rightarrow Hostile(x)$
        \item $American(West)$
        \item $Enemy(Nono,America)$
    \end{enumerate}
    Meta: probar $Criminal(West)$.
\end{example}

\begin{exercise}
    \begin{enumerate}
        \item Usando la misma KB, construya la traza de \textbf{encadenamiento hacia adelante}: parta de los hechos
        (reglas 2, 3, 7, 8) y muestre, en una tabla como la anterior, en qué orden se disparan las reglas 5, 6, 4 y 1
        hasta derivar $Criminal(West)$.
        \item Compare las dos trazas: ¿por qué el encadenamiento hacia atrás solo explora los hechos relevantes para
        la meta, mientras que el hacia adelante puede derivar hechos que nunca se usan?
        \item Dado el hecho adicional $Enemy(Nono, UK)$ (en vez de $America$), ¿sigue siendo posible probar
        $Criminal(West)$ con esta KB? Justifique cuál regla falla.
    \end{enumerate}
\end{exercise}

\begin{exercise}[Resolución en LPO]
    Sean las cláusulas
    \[
        \neg Ama(x,\ Padre(x)) \lor Feliz(x) \qquad \text{y} \qquad Ama(\text{Juan},\ Padre(\text{Juan})).
    \]
    \begin{enumerate}
        \item Encuentre el unificador $\theta$ entre los literales complementarios.
        \item Aplique $\text{SUBST}(\theta, \cdot)$ a la cláusula resolvente resultante.
        \item ¿Qué hecho queda demostrado?
    \end{enumerate}
\end{exercise}
