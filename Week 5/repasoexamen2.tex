\subsection{Probabilidad marginal, conjunta y condicional}

\begin{definition}
    \textbf{Probabilidad conjunta}: P(A,B) es la probabilidad de que ocurran simultaneamente los eventos A y B. Se representa como una tabla 
    donde cada celda es $P=(A=a,B=b)$ y la suma de todas las celdas es 1.
\end{definition}

\begin{definition}
    \textbf{Probabilidad marginal}: Se obtiene sumando sobre la variable que se quiere eliminar:
    \[
        P(A=a) = \sum_{b} P(A=a, B=b)
    \]
\end{definition}

En una tabla, corresponde a sumar la fila o columna correspondiente.

\begin{definition}
    \textbf{Probabilidad condicional}: La fracción de los casos donde ocurre B es los que tambien ocurre A.
    \[
        P(A|B) = \frac{P(A,B)}{P(B)}, P(B)>0
    \]
\end{definition}

Se presente una asimetría importante donde $P(A|B) \neq P(B|A)$ en general. 

\subsection{Regla del producto y regla de la cadena}

\begin{theorem}
    \textbf{Regla del producto}: 
    \[
        P(A,B) = P(A|B)P(B) = P(B|A)P(A)
    \]
\end{theorem}

\begin{theorem}
    \textbf{Regla de la cadena}: 
    \[
        P(x_1, ..., x_n) = P(x_1)P(x_2|x_1)P(x_3|x_1,x_2) ... P(x_n|x_1,...,x_n-1)
    \]
    Sin ningun supuesto de independencia, el numero de parametros libres crece exponencialmente con $n$. Cada supuesto
    de independencia condicional que se agrega elimina variables de la lista de condicionales y reduce ese numero.
\end{theorem}
